AIDA Widget Demo

This page demonstrates how to use the AIDA widget on any website.

Widget Demo

Installation Code


<!-- AIDA Widget -->
<link rel="stylesheet" href="https://testaida.iverse.space/aida-widget.css">
<div id="aida-widget-container"></div>
<script src="https://testaida.iverse.space/aida-widget.umd.js"></script>
<script>
  document.addEventListener('DOMContentLoaded', function() {
    if (window.AidaWidget) {
      AidaWidget.render("#aida-widget-container", {
        language: 'en',
        user: { email: 'user@example.com' },
        translations: {
          transcribing: 'Transcribing...',
          inputPlaceholder: 'Type a message to Aida...'
        }
      });
    }
  });
</script>
